#!/bin/bash
if [ $# -eq 2 ]; then #Se comprueba que el número de parámetros es 2 (opción y dni)
 if [[ $2 =~ ^[0-9]{8}[A-Za-z]$ ]]; then #Se comprueba que el segundo parámetro (dni) tenga el formato correcto
#--------- COGER LOS DATOS DEL TRABAJADOR --------
 Num=$(grep -i "$2" bdreparto | cut -d ':' -f1)
 Nom=$(grep -i "$2" bdreparto | cut -d ':' -f2)
 Apellido=$(grep -i "$2" bdreparto | cut -d ':' -f3)
 DNI=$(grep -i "$2" bdreparto | cut -d ':' -f4)
 SueldoMensual=$(grep -i "$2" bdreparto | cut -d ':' -f5)
 Puesto=$(grep -i "$2" bdreparto | cut -d ':' -f6)
#-------------------------------------------------
if [ "$DNI" != "" ]; then
 case $1 in
 -m)
 echo "NºEmpleado Nombre Apellido DNI Puesto Trabajo Sueldo Mensual(euros)"
 echo "$Num $Nom $Apellido $DNI $Puesto $SueldoMensual"
 ;;
 -a)
 case $Puesto in
 'Productor') extraordinaria=1500 ;;
 'Director') extraordinaria=900 ;;
 'Actor principal') extraordinaria=854 ;;
 'Actor') extraordinaria=700 ;;
 'Guionista') extraordinaria=500 ;;
 esac
 Anual=$(( $SueldoMensual*12 ))
 Extra=$(( $extraordinaria*2 ))
 SueldoAnual=$(( $Anual + $Extra ))
 echo "Nombre Apellido DNI Sueldo Mensual Sueldo Anual"
 echo $Nom $Apellido $DNI $SueldoMensual "euros" $SueldoAnual "euros"
 ;;
 *)
 echo -e "La opción especificada no existe. Las opciones posibles son:\n-m \n-a "
 ;;
 esac
 echo ""
 else
 echo -e "El DNI especificado no existe \n"
 fi
 else
 echo -e "Error: El DNI no tiene el formato correcto. \n"
 fi
else
 echo -e "Debes introducir exactamente 2 parámetros \n"
fi
